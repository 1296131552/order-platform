## 一、快速开始

### 1.1 创建项目
 
```bash
# 使用 Vite 创建 Vue 3 + TypeScript 项目
npm create vite@latest order-platform-web -- --template vue-ts

# 进入目录
cd order-platform-web

# 安装依赖
npm install

# 安装 Element Plus
npm install element-plus @element-plus/icons-vue

# 安装其他依赖
npm install vue-router@4 pinia axios echarts dayjs
npm install @vueuse/core
npm install -D sass
```

```typescript

### 1.4 配置自动导入（推荐）

使用 `unplugin-auto-import` 和 `unplugin-vue-components` 自动导入 Vue API 和 Element Plus 组件：

```typescript
// vite.config.ts
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import AutoImport from 'unplugin-auto-import/vite'
import Components from 'unplugin-vue-components/vite'
import { ElementPlusResolver } from 'unplugin-vue-components/resolvers'

export default defineConfig({
  plugins: [
    vue(),
    // 自动导入 Vue API（ref、computed、reactive 等）
    AutoImport({
      imports: [
        'vue',
        'vue-router',
        'pinia',
        '@vueuse/core'
      ],
      dts: 'src/auto-imports.d.ts',
      resolvers: [ElementPlusResolver()]
    }),
    // 自动导入 Element Plus 组件
    Components({
      dts: 'src/components.d.ts',
      resolvers: [ElementPlusResolver()]
    })
  ]
})
```

**自动导入后无需手动导入**：
- ✅ Vue API（ref、computed、reactive、watch 等）
- ✅ VueRouter API（useRouter、useRoute 等）
- ✅ Pinia API（defineStore、storeToRefs 等）
- ✅ Element Plus 组件
- ✅ VueUse 工具函数

### 3.2 路由守卫（权限控制）

```typescript
// src/router/guards/index.ts
import type { Router } from 'vue-router'
import { ElMessage } from 'element-plus'

export function setupRouterGuards(router: Router) {
  router.beforeEach((to, from, next) => {
    // 设置页面标题
    document.title = `${to.meta.title || '订单可视化平台'} - 订单可视化数字化管理平台`

    // 开发环境：跳过登录验证
    if (import.meta.env.DEV) {
      next()
      return
    }

    // 白名单路由（不需要登录）
    const whiteList = ['/login']

    if (whiteList.includes(to.path)) {
      next()
      return
    }

    // 检查token
    const token = localStorage.getItem('token')
    if (!token) {
      ElMessage.warning('请先登录')
      next('/login')
      return
    }

    next()
  })

  router.afterEach((to, from) => {
    // 可以在这里添加页面访问统计等逻辑
  })
}
```

**开发环境说明**：
- 开发环境（`import.meta.env.DEV`）下自动跳过登录验证
- 生产环境需要真实的 token 验证
- 这样可以方便地进行前端开发，无需等待后端 API
```

### 3.3 路由元信息（meta）

| 字段 | 说明 | 示例 |
|------|------|------|
| title | 页面标题 | `meta: { title: '订单管理' }` |
| icon | 菜单图标 | `meta: { icon: 'Document' }` |
| hideInMenu | 在菜单中隐藏 | `meta: { hideInMenu: true }` |
| hidden | 完全隐藏（面包屑也不显示） | `meta: { hidden: true }` |
| requiresAuth | 是否需要登录 | `meta: { requiresAuth: true }` |

## 四、状态管理（Pinia）

### 4.1 Store 定义规范

使用 **组合式 API 风格**定义 Store：

```typescript
// src/stores/user.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

interface UserInfo {
  id: number
  username: string
  realName: string
  email: string
  avatar: string
  roles: string[]
}

export const useUserStore = defineStore('user', () => {
  // 状态定义
  const userInfo = ref<UserInfo | null>(null)

  // 计算属性
  const isLoggedIn = computed(() => !!userInfo.value)

  // 操作方法
  function setUserInfo(info: UserInfo) {
    userInfo.value = info
  }

  function clearUserInfo() {
    userInfo.value = null
  }

  // 返回暴露的属性和方法
  return {
    userInfo,
    setUserInfo,
    clearUserInfo,
    isLoggedIn
  }
})
```

### 4.2 应用状态 Store

```typescript
// src/stores/app.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const useAppStore = defineStore('app', () => {
  // 侧边栏折叠状态
  const isCollapse = ref(false)

  // 深色主题
  const isDark = ref(false)

  // 加载状态
  const loading = ref(false)

  // 侧边栏宽度（计算属性）
  const sidebarWidth = computed(() => {
    return isCollapse.value ? 64 : 200
  })

  // 是否为移动端
  const isMobile = computed(() => {
    return window.innerWidth < 768
  })

  // 方法
  function toggleCollapse() {
    isCollapse.value = !isCollapse.value
  }

  function toggleTheme() {
    isDark.value = !isDark.value
  }

  function setLoading(loading: boolean) {
    loading.value = loading
  }

  return {
    isCollapse,
    isDark,
    loading,
    sidebarWidth,
    toggleCollapse,
    toggleTheme,
    setLoading,
    isMobile
  }
})
```

### 5.3 头部组件

```vue
<!-- src/components/layout/Header.vue -->
<template>
  <div class="header-container">
    <div class="left">
      <!-- 折叠按钮 -->
      <el-icon class="collapse-icon" @click="toggleCollapse">
        <Fold v-if="!isCollapse" />
        <Expand v-else />
      </el-icon>
      <!-- 面包屑 -->
      <el-breadcrumb separator="/">
        <el-breadcrumb-item :to="{ path: '/dashboard' }">首页</el-breadcrumb-item>
        <el-breadcrumb-item v-if="currentRoute.meta.title">
          {{ currentRoute.meta.title }}
        </el-breadcrumb-item>
      </el-breadcrumb>
    </div>
    <div class="right">
      <!-- 用户菜单 -->
      <el-dropdown @command="handleCommand">
        <span class="user-info">
          <el-avatar :size="32" :src="userAvatar" />
          <span class="username">{{ userName }}</span>
        </span>
        <template #dropdown>
          <el-dropdown-menu>
            <el-dropdown-item command="profile">个人中心</el-dropdown-item>
            <el-dropdown-item command="settings">系统设置</el-dropdown-item>
            <el-dropdown-item divided command="logout">退出登录</el-dropdown-item>
          </el-dropdown-menu>
        </template>
      </el-dropdown>
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import { useRouter, useRoute } from 'vue-router'
import { ElMessage } from 'element-plus'
import { Fold, Expand } from '@element-plus/icons-vue'
import { useAppStore } from '@/stores/app'
import { useUserStore } from '@/stores/user'

const router = useRouter()
const route = useRoute()
const appStore = useAppStore()
const userStore = useUserStore()

const isCollapse = computed(() => appStore.isCollapse)
const currentRoute = computed(() => route)
const userName = computed(() => userStore.userInfo?.realName || '管理员')
const userAvatar = computed(() => userStore.userInfo?.avatar || '')

function toggleCollapse() {
  appStore.toggleCollapse()
}

function handleCommand(command: string) {
  switch (command) {
    case 'profile':
      ElMessage.info('个人中心功能开发中')
      break
    case 'settings':
      router.push('/system')
      break
    case 'logout':
      userStore.clearUserInfo()
      localStorage.removeItem('token')
      ElMessage.success('退出登录成功')
      router.push('/login')
      break
  }
}
</script>
```

## 九、模拟数据降级处理

### 9.1 为什么需要模拟数据降级

在开发阶段，后端 API 可能还未完成或无法访问。为了让前端开发不阻塞，我们需要实现模拟数据降级机制：

- API 调用失败时，自动使用模拟数据
- 不影响开发进度和功能演示
- 后续对接真实 API 时只需移除降级逻辑

### 9.2 实现方式

```typescript
// 在列表页面的 loadData 方法中实现
const mockData = [
  { id: 1, name: '示例数据1', status: 'active' },
  { id: 2, name: '示例数据2', status: 'inactive' }
]

const loadData = async () => {
  loading.value = true
  try {
    const params = { ...searchForm.value, page: page.value, pageSize: pageSize.value }
    const res = await getSupplierList(params)
    tableData.value = res.list
    total.value = res.total
  } catch (error) {
    console.error('加载列表失败：', error)
    // API 失败时使用模拟数据
    tableData.value = mockData
    total.value = mockData.length
  } finally {
    loading.value = false
  }
}
```

### 9.3 最佳实践

1. **模拟数据结构应与 API 返回格式一致**
2. **在控制台输出清晰的错误提示**（方便调试）
3. **生产环境移除模拟数据降级逻辑**

```typescript
const loadData = async () => {
  loading.value = true
  try {
    const res = await getDataList()
    tableData.value = res.list
    total.value = res.total
  } catch (error) {
    console.error('加载数据失败：', error)

    // 仅开发环境使用模拟数据
    if (import.meta.env.DEV) {
      tableData.value = mockData
      total.value = mockData.length
    } else {
      ElMessage.error('数据加载失败，请稍后重试')
    }
  } finally {
    loading.value = false
  }
}
```

function handleDelete() {
  confirmDelete(() => {
    // 确认后的回调
    console.log('删除成功')
  })
}
</script>

<template>
  <el-dialog v-model="visible" :title="title">
    {{ message }}
    <template #footer>
      <el-button @click="handleCancel">取消</el-button>
      <el-button type="primary" @click="handleConfirm">确认</el-button>
    </template>
  </el-dialog>
</template>
```

## 十一、API 接口封装

### 11.1 Axios 实例配置

```typescript
// src/api/index.ts
import axios from 'axios'
import { ElMessage } from 'element-plus'

const request = axios.create({
  baseURL: '/api',
  timeout: 10000
})

// 请求拦截器
request.interceptors.request.use(
  config => {
    const token = localStorage.getItem('token')
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  error => {
    return Promise.reject(error)
  }
)

// 响应拦截器
request.interceptors.response.use(
  response => {
    return response.data
  },
  error => {
    ElMessage.error(error.message)
    return Promise.reject(error)
  }
)

export default request
```

### 11.2 业务 API 封装

```typescript
// src/api/order.ts
import request from './index'

export const orderApi = {
  // 订单列表
  list: (params: any) => {
    return request.get('/orders', { params })
  },

  // 订单详情
  detail: (id: number) => {
    return request.get(`/orders/${id}`)
  },

  // 创建订单
  create: (data: any) => {
    return request.post('/orders', data)
  },

  // 更新订单
  update: (id: number, data: any) => {
    return request.put(`/orders/${id}`, data)
  },

  // 删除订单
  delete: (id: number) => {
    return request.delete(`/orders/${id}`)
  }
}
```

## 十四、最佳实践

### 14.1 组件拆分原则

- **可复用**：超过2次使用的功能抽取为组件
- **单一职责**：一个组件只做一件事
- **Props向下**：父组件通过 props 传递数据
- **Events向上**：子组件通过 emit 传递事件

### 14.2 性能优化

- **使用 `v-show` 而非 `v-if`**：频繁切换的场景
- **合理使用 `computed`**：缓存计算结果
- **避免不必要的响应式**：使用 `shallowRef`、`shallowReactive`

### 14.3 代码规范

- **使用 TypeScript**：提供类型安全
- **组件命名清晰**：见名知意
- **添加注释**：复杂逻辑必须注释
- **统一代码风格**：使用 ESLint + Prettier

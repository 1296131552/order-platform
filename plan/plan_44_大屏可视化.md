## 核心接口定义

### 主要类/接口
```typescript
export const dashboardApi = {
  // KPI指标
  kpi: (params: KpiRequest) => request.get<KpiVO>('/api/dashboard/kpi', { params }),
  // 趋势数据
  trend: (params: TrendRequest) => request.get<TrendVO>('/api/dashboard/trend', { params }),
  // 排行榜
  ranking: (params: RankingRequest) => request.get<RankingVO>('/api/dashboard/ranking', { params }),
  // 地图数据
  mapData: (params: MapRequest) => request.get<MapVO>('/api/visualization/map', { params })
};
```
